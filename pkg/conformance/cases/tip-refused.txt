A tooltip is a title= or one <title>, not both, and a <title> goes in a
widget. Nothing in one acts, since it goes when the pointer moves: not
in it, nor in a partial it uses. A partial's use takes title= as a prop.
-- form.surf --
<surface>
  <text title="Plain">x<title>Rich</title></text>
  <row><title>One</title><title>Two</title></row>
  <row>
    <title>
      <button>Go</button>
      <text @click="add(1)">x</text>
      <stack focusable/>
      <field/>
      <row><menu><action do="add(1)">Add</action></menu></row>
      <text title="Inner">y</text>
      <bind key="a" do="add(1)"/>
    </title>
  </row>
  <title>Stray</title>
  <row><title :if="true">Has a directive</title></row>
  <tip-part title="A prop"/>
  <row><title><tip-part/></title></row>
  <tip-part><title>In a use</title></tip-part>
</surface>
-- tip-part.surf --
<props title="string"/>
<row :title="title"><slot/><button @click="add(2)">Go</button></row>
